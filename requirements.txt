# Direct dependencies only, pinned to the versions this project was built and
# tested against. Deliberately not a `pip freeze` dump: freezing records every
# transitive package, which makes it impossible to see what the project itself
# actually depends on and breaks whenever a sub-dependency is repinned.

# Web API. python-multipart is not imported by our code but FastAPI requires it
# to parse the multipart body /api/upload receives; without it the endpoint
# raises at import time, so it is a direct dependency in every sense that matters.
fastapi==0.141.1
uvicorn==0.52.4
python-multipart==0.0.32
pydantic==2.13.4

# app.py only -- the Streamlit dev harness, kept as a second front end over the
# same core/. It is not the deliverable; backend/ + frontend/ are.
streamlit==1.62.0

plotly==6.9.0
pandas==3.0.5
numpy==2.5.2
scikit-learn==1.9.0
python-dotenv==1.2.3

# LLM access, all of it behind core/llm.py.
#
# httpx is the DEFAULT provider's entire client. DeepSeek speaks the
# OpenAI-compatible chat-completions protocol, which is one POST with four
# fields, so the `openai` package would be a dependency bought for nothing.
httpx==0.28.1

# google-genai is only needed for AI_PROVIDER=gemini. core/llm.py imports it
# defensively and reports the absence rather than raising, so a DeepSeek-only
# deployment can drop this line and the app still starts.
google-genai==2.19.0
